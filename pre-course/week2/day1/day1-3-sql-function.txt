COUNT: 검색한 결과의 데이터의 개수를 가져오는 함수 ( NULL은 제외 )
SELECT COUNT(칼럼) FROM 테이블;
SELECT COUNT(*) FROM book;

LIMIT: 데이터의 일부만 출력하도록 제한하는 함수
SELECT * FROM book LIMIT 5;
SELECT * FROM book LIMIT 1, 5;
- 칼럼의 시작은 0, 2번째 데이터부터 5개를 가져오라는 의미

SUM: 지정한 컬럼들의 값을 모두 더해 돌려주는 함수
SELECT SUM(칼럼) FROM 테이블;

AVG: 지정한 칼럼들의 평균값을 구해주는 내장함수
SELECT AVG(칼럼) FROM 테이블;

MAX: 테이블에서 가장 높은 값을 찾음 (문자형도 가능)
SELECT MAX(칼럼) FROM 테이블;

MIN: 가장 낮은 값은 찾음
SELET MIN(칼럼) FROM 테이블;
